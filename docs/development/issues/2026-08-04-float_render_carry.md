# Render an f64 into buf with `decimals` fraction digits, propagating the
# rounding CARRY out of the fractional digits into the integer part.
#
# stdlib fmt_float_buf (lib/fmt.cyr:246-258) computes
# frac = round(|val - floor(val)| * 10^decimals) and then zero-pads under
# `if (pad > 0)` with pad = decimals - flen. When the fraction rounds UP to
# exactly 10^decimals it needs decimals+1 digits, pad goes negative, the pad
# branch is skipped, and the overflowed digit is emitted AFTER the '.' with
# nothing carrying into the whole part: 0.9999999 rendered as "0.1000000" and
# 2.9999999 as "2.1000000" (audit 2026-08-04). We cannot patch lib/ -- it is
# regenerated by `cyrius lib sync` -- so hisab carries its own renderer. It is
# the stdlib algorithm plus the two lines that fold the overflow into `whole`.
# Deliberately NOT fixed by pre-rounding val to `decimals` places: val * 10^6
# is inexact above ~1e15 and would perturb the large-magnitude integral values
# that _latex_fmt_const routes down this same branch.
fn _sym_render_f64(val, buf, decimals) {
    # inf/nan have no whole/fraction split -- the stdlib has its own path.
    if (((val >> 52) & 0x7FF) == 0x7FF) {
        return fmt_float_buf(val, buf, decimals);
    }
    var neg = 0;
    var mag = val;
    if (((val >> 63) & 1) == 1) { neg = 1; mag = f64_abs(val); }
    var whole = f64_to(f64_floor(mag));
    var scale = F64_ONE;
    var scale_i = 1;
    var di = 0;
    while (di < decimals) {
        scale = f64_mul(scale, f64_from(10));
        scale_i = scale_i * 10;
        di = di + 1;
    }
    var frac = f64_to(f64_round(f64_mul(f64_sub(mag, f64_from(whole)), scale)));
    if (frac < 0) { frac = 0; }
    # THE CARRY the stdlib drops.
    if (frac >= scale_i) {
        frac = frac - scale_i;
        whole = whole + 1;
    }
    var pos = 0;
    if (neg == 1) { store8(buf, 45); pos = 1; }
    pos = pos + fmt_int_buf(whole, buf + pos);
    store8(buf + pos, 46);
    pos = pos + 1;
    var flen = fmt_int_buf(frac, buf + pos);
    var pad = decimals - flen;
    if (pad > 0) {
        var si = flen - 1;
        while (si >= 0) {
            store8(buf + pos + pad + si, load8(buf + pos + si));
            si = si - 1;
        }
        si = 0;
        while (si < pad) { store8(buf + pos + si, 48); si = si + 1; }
        flen = decimals;
    }
    return pos + flen;
}
